# Home Assistant integration test harness.
#
# Pinned, because the harness pins Home Assistant, which pins habluetooth and
# cryptography -- so an unpinned install resolves differently on every machine
# and the suite stops being reproducible. This is the version the tests are
# written against; the integration itself is verified on much newer Home
# Assistant releases against real hardware (decisions.md D-010, D-011).
pytest-homeassistant-custom-component==0.13.205

# Runtime dependencies from the integration's manifest.
bthome-ble>=3.22.1
bleak-retry-connector>=3.5.0

# Home Assistant's `bluetooth` component imports its `usb` component, which
# imports pyserial. Real deployments get these from Home Assistant's own
# bootstrap requirements; the test harness does not pull them, so the import
# fails at collection time with a message that names neither Bluetooth nor USB.
pyserial
pyserial-asyncio-fast
# ... and on Linux that component also wants pyudev, which is why this suite
# passed on a Windows workstation and failed in CI.
pyudev; sys_platform == "linux"
